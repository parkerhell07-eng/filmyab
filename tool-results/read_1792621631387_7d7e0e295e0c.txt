    26→    26→/**
    27→    27→ * FilmYab v2 on-device SMART ENGINE (no server, no keys, no settings).
    28→    28→ *
    29→    29→ * Pipeline:
    30→    30→ *  1. net     — automatic VPN/connectivity check (Instagram + internet reachability)
    31→    31→ *  2. fetch   — scrape the Instagram post on device (crawler-UA page, embed, __a=1, og)
    32→    32→ *  3. analyze — download the video to cache, sample 3 frames (MediaMetadataRetriever),
    33→    33→ *               read on-screen text with on-device OCR (ML Kit, bundled latin model)
    34→    34→ *  4. match   — keyless text LLM (Pollinations) turns caption/hashtags/OCR/og-title
    35→    35→ *               into ranked movie candidates
    36→    36→ *  5. verify  — each candidate is verified against Wikipedia REST (keyless): title,
    37→    37→ *               year, director, description, poster
    38→    38→ *  6. final   — self-built scoring blends LLM confidence with concrete evidence
    39→    39→ *               (wiki confirmation, og-title hit, hashtag hit, OCR hit); a second
    40→    40→ *               LLM pass fuses everything when the score is not decisive.
    41→    41→ */

    93→    93→    suspend fun checkNetwork(): NetState = withContext(Dispatchers.IO) {
    94→    94→        val instagram = reachable("https://www.instagram.com/robots.txt")
    95→    95→        val internet = instagram ||
    96→    96→            reachable("https://www.gstatic.com/generate_204") ||
    97→    97→            reachable("https://en.wikipedia.org/w/rest.php/v1/search/title?q=a&limit=1")
    98→    98→        when {
    99→    99→            instagram -> NetState("ok", instagramOk = true, internetOk = true)
   100→   100→            internet -> NetState("vpn_needed", instagramOk = false, internetOk = true)
   101→   101→            else -> NetState("offline", instagramOk = false, internetOk = false)
   102→   102→        }
   103→   103→    }

   116→   116→
   117→   117→    suspend fun identify(
   118→   118→        appContext: Context,
   119→   119→        rawUrl: String,
   120→   120→        onStep: (String, String) -> Unit
   121→   121→    ): Outcome = withContext(Dispatchers.IO) {
   122→   122→
   123→   123→        /* ---- 1. net ---- */
   124→   124→        onStep(STEP_NET, "start")
   125→   125→        val net = checkNetwork()
   126→   126→        onStep(STEP_NET, if (net.internetOk || net.instagramOk) "done" else "error")
   127→   127→        if (!net.internetOk) {
   128→   128→            return@withContext Outcome.Fail("اینترنت در دسترس نیست. اتصال شبکه یا فیلترشکن را بررسی کنید.")
   129→   129→        }
   130→   130→
   131→   131→        /* ---- 2. fetch ---- */
   132→   132→        onStep(STEP_FETCH, "start")
   133→   133→        val post = InstagramScraper.fetchPost(rawUrl)
   134→   134→        if (post == null) {
   135→   135→            onStep(STEP_FETCH, "error")
   136→   136→            return@withContext Outcome.Fail(
   137→   137→                "اینستاگرام اجازه خواندن این پست را نداد. اگر فیلترشکن داری روشنش کن، چند لحظه صبر کن و دوباره تلاش کن. (پست باید عمومی باشد)",
   138→   138→                needVpn = !net.instagramOk
   139→   139→            )
   140→   140→        }
   141→   141→        onStep(STEP_FETCH, "done")
   142→   142→
   143→   143→        /* ---- 3. analyze: frames + OCR ---- */
   144→   144→        onStep(STEP_ANALYZE, "start")
   145→   145→        var ocrText = ""
   146→   146→        val videoUrl = post.videoUrl
   147→   147→        if (videoUrl != null) {
   148→   148→            val frames = extractFrames(appContext, videoUrl, 3)
   149→   149→            if (frames.isNotEmpty()) {
   150→   150→                ocrText = runOcr(frames)
   151→   151→                onStep(STEP_ANALYZE, "done")
   152→   152→            } else {
   153→   153→                onStep(STEP_ANALYZE, "skip")
   154→   154→            }
   155→   155→        } else {
   156→   156→            onStep(STEP_ANALYZE, "skip")
   157→   157→        }
   158→   158→
   159→   159→        /* ---- 4. match: LLM candidates ---- */
   160→   160→        onStep(STEP_MATCH, "start")
   161→   161→        val ai = askLlmCandidates(post, ocrText)
   162→   162→        if (ai == null) {
   163→   163→            onStep(STEP_MATCH, "error")
   164→   164→            return@withContext Outcome.Fail("سرویس تحلیل هوشمند پاسخ نداد. چند لحظه بعد دوباره امتحان کن.")
   165→   165→        }
   166→   166→        if (ai.candidates.isEmpty()) {
   167→   167→            onStep(STEP_MATCH, "error")
   168→   168→            val extra = ai.sceneSummary?.let { " توضیح صحنه: $it" } ?: ""
   169→   169→            return@withContext Outcome.Fail(
   170→   170→                "با شواهد موجود (کپشن و متن روی تصویر) نتونستم فیلمی رو با اطمینان پیدا کنم.$extra"
   171→   171→            )
   172→   172→        }
   173→   173→        onStep(STEP_MATCH, "done")
   174→   174→
   175→   175→        /* ---- 5. verify: Wikipedia ---- */
   176→   176→        onStep(STEP_VERIFY, "start")
   177→   177→        val verified = verifyCandidates(ai.candidates, post, ocrText)
   178→   178→        if (verified.isEmpty()) {
   179→   179→            onStep(STEP_VERIFY, "error")
   180→   180→            val names = ai.candidates.joinToString("، ") { it.name ?: "?" }
   181→   181→            return@withContext Outcome.Fail(
   182→   182→                "کاندیدها ($names) در پایگاه فیلم تأیید نشدند. احتمالاً کلیپ از فیلم شناخته‌شده‌ای نیست یا شواهد کم است."
   183→   183→            )
   184→   184→        }
   185→   185→        onStep(STEP_VERIFY, "done")
   186→   186→
   187→   187→        /* ---- 6. final ---- */
   188→   188→        onStep(STEP_FINAL, "start")
   189→   189→        val best = verified.first()
   190→   190→        var result = buildFinal(best, verified.drop(1), ai, post, ocrText)
   191→   191→        if (best.score < 0.85) {
   192→   192→            val fused = askLlmFusion(verified, ai, post, ocrText)
   193→   193→            if (fused != null) {
   194→   194→                result = blend(result, fused)
   195→   195→            }
   196→   196→        }
   197→   197→        onStep(STEP_FINAL, "done")
   198→   198→        Outcome.Done(result)
   199→   199→    }

   203→   203→    private fun extractFrames(context: Context, videoUrl: String, count: Int): List<Bitmap> {
   204→   204→        var file: File? = null
   205→   205→        var mmr: MediaMetadataRetriever? = null
   206→   206→        try {
   207→   207→            file = downloadToCache(context, videoUrl) ?: return emptyList()
   208→   208→            mmr = MediaMetadataRetriever()
   209→   209→            mmr.setDataSource(file.absolutePath)
   210→   210→            val durMs = mmr.extractMetadata(MediaMetadataRetriever.METADATA_KEY_DURATION)
   211→   211→                ?.toLongOrNull() ?: 0L
   212→   212→            if (durMs <= 0) return emptyList()
   213→   213→            val out = mutableListOf<Bitmap>()
   214→   214→            val fracs = doubleArrayOf(0.12, 0.45, 0.78)
   215→   215→            for (f in fracs) {
   216→   216→                if (out.size >= count) break
   217→   217→                val bmp = try {
   218→   218→                    mmr.getFrameAtTime(
   219→   219→                        (durMs * 1000.0 * f).toLong(),
   220→   220→                        MediaMetadataRetriever.OPTION_CLOSEST_SYNC
   221→   221→                    )
   222→   222→                } catch (e: Exception) {
   223→   223→                    null
   224→   224→                }
   225→   225→                if (bmp != null) out.add(bmp)
   226→   226→            }
   227→   227→            return out
   228→   228→        } catch (e: Exception) {
   229→   229→            return emptyList()
   230→   230→        } finally {
   231→   231→            try {
   232→   232→                mmr?.release()
   233→   233→            } catch (e: Exception) {
   234→   234→            }
   235→   235→            file?.delete()
   236→   236→        }
   237→   237→    }

   239→   239→    /** Streams the video into cache with a hard size cap. Returns null on any failure. */
   240→   240→    private fun downloadToCache(context: Context, url: String): File? {
   241→   241→        var file: File? = null
   242→   242→        try {
   243→   243→            val req = Request.Builder()
   244→   244→                .url(url)
   245→   245→                .header("User-Agent", BROWSER_UA)
   246→   246→                .header("Referer", "https://www.instagram.com/")
   247→   247→                .build()
   248→   248→            http.newCall(req).execute().use { resp ->
   249→   249→                if (!resp.isSuccessful) return null
   250→   250→                val body = resp.body ?: return null
   251→   251→                val f = File(context.cacheDir, "filmyab_video_${System.currentTimeMillis()}.mp4")
   252→   252→                file = f
   253→   253→                val ok = body.byteStream().use { input ->
   254→   254→                    f.outputStream().use { output -> copyCapped(input, output, MAX_VIDEO_BYTES) }
   255→   255→                }
   256→   256→                if (!ok || f.length() < 10_000) {
   257→   257→                    f.delete()
   258→   258→                    file = null
   259→   259→                    return null
   260→   260→                }
   261→   261→                return f
   262→   262→            }
   263→   263→        } catch (e: Exception) {
   264→   264→            file?.delete()
   265→   265→            return null
   266→   266→        }
   267→   267→    }

   269→   269→    /** Copies with a byte cap; false when the stream exceeded the cap or failed. */
   270→   270→    private fun copyCapped(input: java.io.InputStream, output: java.io.OutputStream, cap: Long): Boolean {
   271→   271→        val buf = ByteArray(64 * 1024)
   272→   272→        var total = 0L
   273→   273→        try {
   274→   274→            while (true) {
   275→   275→                val n = input.read(buf)
   276→   276→                if (n < 0) break
   277→   277→                total += n
   278→   278→                if (total > cap) return false
   279→   279→                output.write(buf, 0, n)
   280→   280→            }
   281→   281→            return true
   282→   282→        } catch (e: Exception) {
   283→   283→            return false
   284→   284→        }
   285→   285→    }

   287→   287→    /** On-device OCR (bundled latin model — no downloads). Returns merged unique lines. */
   288→   288→    private suspend fun runOcr(frames: List<Bitmap>): String = withContext(Dispatchers.IO) {
   289→   289→        val latin = TextRecognition.getClient(TextRecognizerOptions.DEFAULT_OPTIONS)
   290→   290→        val lines = LinkedHashSet<String>()
   291→   291→        try {
   292→   292→            for (bmp in frames) {
   293→   293→                try {
   294→   294→                    val text: Text = latin.process(InputImage.fromBitmap(bmp, 0)).await()
   295→   295→                    for (line in text.text.split("\n")) {
   296→   296→                        val t = line.trim()
   297→   297→                        if (t.length >= 2) {
   298→   298→                            lines.add(t)
   299→   299→                        }
   300→   300→                    }
   301→   301→                } catch (e: Exception) {
   302→   302→                    // OCR of this frame failed — keep going
   303→   303→                }
   304→   304→            }
   305→   305→        } finally {
   306→   306→            try {
   307→   307→                latin.close()
   308→   308→            } catch (e: Exception) {
   309→   309→            }
   310→   310→            frames.forEach { it.recycle() }
   311→   311→        }
   312→   312→        lines.take(14).joinToString(" | ").take(1400)
   313→   313→    }

   317→   317→    private fun callLlm(system: String, user: String): String? {
   318→   318→        val delaysMs = longArrayOf(0, 3000, 8000)
   319→   319→        for (attempt in delaysMs.indices) {
   320→   320→            if (delaysMs[attempt] > 0) {
   321→   321→                try {
   322→   322→                    Thread.sleep(delaysMs[attempt])
   323→   323→                } catch (e: InterruptedException) {
   324→   324→                    Thread.currentThread().interrupt()
   325→   325→                }
   326→   326→            }
   327→   327→            try {
   328→   328→                val payload = JsonObject()
   329→   329→                payload.addProperty("model", "openai")
   330→   330→                val sys = JsonObject()
   331→   331→                sys.addProperty("role", "system")
   332→   332→                sys.addProperty("content", system)
   333→   333→                val usr = JsonObject()
   334→   334→                usr.addProperty("role", "user")
   335→   335→                usr.addProperty("content", user)
   336→   336→                val messages = JsonArray()
   337→   337→                messages.add(sys)
   338→   338→                messages.add(usr)
   339→   339→                payload.add("messages", messages)
   340→   340→                payload.addProperty("temperature", 0.2)
   341→   341→
   342→   342→                val request = Request.Builder()
   343→   343→                    .url(POLLINATIONS_URL)
   344→   344→                    .header("Accept", "application/json")
   345→   345→                    .post(payload.toString().toRequestBody("application/json".toMediaType()))
   346→   346→                    .build()
   347→   347→                http.newCall(request).execute().use { resp ->
   348→   348→                    if (!resp.isSuccessful) throw IOException("HTTP ${resp.code}")
   349→   349→                    val bodyStr = resp.body?.string() ?: throw IOException("empty body")
   350→   350→                    val root = JsonParser.parseString(bodyStr).asJsonObject
   351→   351→                    val choices = root.getAsJsonArray("choices") ?: throw IOException("no choices")
   352→   352→                    if (choices.size() == 0) throw IOException("empty choices")
   353→   353→                    val message = choices.get(0).asJsonObject.getAsJsonObject("message")
   354→   354→                        ?: throw IOException("no message")
   355→   355→                    val content = message.get("content")
   356→   356→                    if (content == null || content.isJsonNull) throw IOException("no content")
   357→   357→                    return content.asString
   358→   358→                }
   359→   359→            } catch (e: IOException) {
   360→   360→                // network/HTTP failure — retry with backoff
   361→   361→            } catch (e: Exception) {
   362→   362→                break
   363→   363→            }
   364→   364→        }
   365→   365→        return null
   366→   366→    }

   368→   368→    private fun extractJson(content: String): JsonObject? {
   369→   369→        var cleaned = content.trim()
   370→   370→        if (cleaned.startsWith("```")) {
   371→   371→            cleaned = cleaned.removePrefix("```json").removePrefix("```").trim()
   372→   372→            cleaned = cleaned.removeSuffix("```").trim()
   373→   373→        }
   374→   374→        val start = cleaned.indexOf('{')
   375→   375→        val end = cleaned.lastIndexOf('}')
   376→   376→        if (start < 0 || end <= start) return null
   377→   377→        return try {
   378→   378→            JsonParser.parseString(cleaned.substring(start, end + 1)).asJsonObject
   379→   379→        } catch (e: Exception) {
   380→   380→            null
   381→   381→        }
   382→   382→    }

   430→   430→    private fun askLlmCandidates(post: InstagramPost, ocrText: String): AiAnalysis? {
   431→   431→        val sb = StringBuilder()
   432→   432→        sb.append("Identify the movie from this Instagram post.\n")
   433→   433→        post.ownerUsername?.let { sb.append("Uploader: @").append(it).append("\n") }
   434→   434→        post.ogTitle?.let { sb.append("Page title: ").append(it.take(200)).append("\n") }
   435→   435→        sb.append("Caption: ").append(post.caption?.take(900) ?: "(none)").append("\n")
   436→   436→        if (post.hashtags.isNotEmpty()) {
   437→   437→            sb.append("Hashtags: ").append(post.hashtags.joinToString(" ")).append("\n")
   438→   438→        }
   439→   439→        if (ocrText.isNotBlank()) {
   440→   440→            sb.append("On-screen text (OCR): ").append(ocrText).append("\n")
   441→   441→        }
   442→   442→        val content = callLlm(CANDIDATES_SYSTEM, sb.toString()) ?: return null
   443→   443→        val obj = extractJson(content) ?: return null
   444→   444→
   445→   445→        val candidates = mutableListOf<Candidate>()
   446→   446→        val arr = obj.get("candidates")
   447→   447→        if (arr != null && arr.isJsonArray) {
   448→   448→            for (el in arr.asJsonArray) {
   449→   449→                if (candidates.size >= 3) break
   450→   450→                val o = el.takeIf { it.isJsonObject }?.asJsonObject ?: continue
   451→   451→                val title = optString(o, "title") ?: continue
   452→   452→                candidates.add(
   453→   453→                    Candidate(
   454→   454→                        name = title,
   455→   455→                        year = optString(o, "year"),
   456→   456→                        confidence = optDouble(o, "confidence").coerceIn(0.0, 1.0),
   457→   457→                        reason = optString(o, "reason_fa")
   458→   458→                    )
   459→   459→                )
   460→   460→            }
   461→   461→        }
   462→   462→        return AiAnalysis(
   463→   463→            found = optBool(obj, "found") && candidates.isNotEmpty(),
   464→   464→            sceneSummary = optString(obj, "scene_summary_fa"),
   465→   465→            candidates = candidates
   466→   466→        )
   467→   467→    }

   477→   477→    private fun wikiSearch(lang: String, query: String, limit: Int): List<WikiPage> {
   478→   478→        return try {
   479→   479→            val q = URLEncoder.encode(query, "UTF-8")
   480→   480→            val req = Request.Builder()
   481→   481→                .url("https://$lang.wikipedia.org/w/rest.php/v1/search/title?q=$q&limit=$limit")
   482→   482→                .header("User-Agent", BROWSER_UA)
   483→   483→                .header("Accept", "application/json")
   484→   484→                .build()
   485→   485→            http.newCall(req).execute().use { resp ->
   486→   486→                if (!resp.isSuccessful) return@use emptyList()
   487→   487→                val bodyStr = resp.body?.string() ?: return@use emptyList()
   488→   488→                val root = JsonParser.parseString(bodyStr).asJsonObject
   489→   489→                val pages = root.getAsJsonArray("pages") ?: return@use emptyList()
   490→   490→                val out = mutableListOf<WikiPage>()
   491→   491→                for (el in pages) {
   492→   492→                    val o = el.takeIf { it.isJsonObject }?.asJsonObject ?: continue
   493→   493→                    val key = optString(o, "key") ?: continue
   494→   494→                    val title = optString(o, "title") ?: key
   495→   495→                    out.add(WikiPage(key, title, optString(o, "description")))
   496→   496→                }
   497→   497→                out
   498→   498→            }
   499→   499→        } catch (e: Exception) {
   500→   500→            emptyList()
   501→   501→        }
   502→   502→    }

   504→   504→    private fun wikiSummary(lang: String, key: String): JsonObject? {

   506→   506→            val encoded = URLEncoder.encode(key, "UTF-8").replace("+", "%20")
   507→   507→            val req = Request.Builder()
   508→   508→                .url("https://$lang.wikipedia.org/api/rest_v1/page/summary/$encoded")

   511→   511→                .build()
   512→   512→            http.newCall(req).execute().use { resp ->
   513→   513→                if (!resp.isSuccessful) return@use null
   514→   514→                val bodyStr = resp.body?.string() ?: return@use null

   522→   522→    private val filmDescRegex = Regex("""(?i)\b(film|movie|series|television|miniseries)\b""")
   523→   523→    private val yearRegex = Regex("""(19|20)\d{2}""")
   524→   524→    private val directorRegex = Regex("""\bby ([A-Z][^,.;()]{2,40})""")
   525→   525→
   526→   526→    private fun buildWikiEvidence(page: WikiPage): WikiEvidence? {
   527→   527→        val summary = wikiSummary("en", page.key) ?: return null
   528→   528→        val description = optString(summary, "description")
   529→   529→        val extract = optString(summary, "extract") ?: ""
   530→   530→        val hay = listOfNotNull(description, extract.take(400)).joinToString(" ")
   531→   531→        val isFilm = filmDescRegex.containsMatchIn(hay) ||
   532→   532→            hay.contains("فیلم") || hay.contains("سریال")
   533→   533→        if (!isFilm) return null
   534→   534→
   535→   535→        val year = description?.let { yearRegex.find(it)?.value }
   536→   536→            ?: yearRegex.find(extract.take(200))?.value
   537→   537→        val director = description?.let { directorRegex.find(it)?.groupValues?.get(1)?.trim() }
   538→   538→        val originalImage = summary.getAsJsonObject("originalimage")
   539→   539→        val thumbnailImage = summary.getAsJsonObject("thumbnail")
   540→   540→        val poster = originalImage?.optStringSafe("source")
   541→   541→            ?: thumbnailImage?.optStringSafe("source")
   542→   542→        val pageUrl = try {
   543→   543→            summary.getAsJsonObject("content_urls")
   544→   544→                ?.getAsJsonObject("desktop")?.optStringSafe("page")
   545→   545→        } catch (e: Exception) {
   546→   546→            null
   547→   547→        }
   548→   548→
   549→   549→        // Persian title: search fa-wiki with the english title, accept entries about films
   550→   550→        var faTitle: String? = null
   551→   551→        val faPages = wikiSearch("fa", page.title, 3)
   552→   552→        for (fp in faPages) {
   553→   553→            val d = fp.description ?: ""
   554→   554→            if (d.contains("فیلم") || d.contains("سریال")) {
   555→   555→                faTitle = fp.title
   556→   556→                break
   557→   557→            }
   558→   558→        }
   559→   559→
   560→   560→        return WikiEvidence(
   561→   561→            pageTitle = page.title,
   562→   562→            pageDescription = description,
   563→   563→            year = year,
   564→   564→            director = director,
   565→   565→            pageUrl = pageUrl,
   566→   566→            posterUrl = poster,
   567→   567→            faTitle = faTitle,
   568→   568→            isFilm = true
   569→   569→        )
   570→   570→    }
   571→   571→
   572→   572→    private fun verifyCandidates(
   573→   573→        candidates: List<Candidate>,
   574→   574→        post: InstagramPost,
   575→   575→        ocrText: String
   576→   576→    ): List<VerifiedCandidate> {
   577→   577→        val out = mutableListOf<VerifiedCandidate>()
   578→   578→        val seen = mutableSetOf<String>()
   579→   579→        for (c in candidates) {
   580→   580→            if (out.size >= 3) break
   581→   581→            val title = c.name ?: continue
   582→   582→            val key = normalizedKey(title)
   583→   583→            if (key.isEmpty() || seen.contains(key)) continue
   584→   584→            seen.add(key)
   585→   585→
   586→   586→            val pages = wikiSearch("en", title, 5)
   587→   587→            var chosen: WikiPage? = null
   588→   588→            for (p in pages) {
   589→   589→                val d = p.description ?: ""
   590→   590→                if (filmDescRegex.containsMatchIn(d) || d.contains("فیلم")) {
   591→   591→                    chosen = p
   592→   592→                    break
   593→   593→                }
   594→   594→            }
   595→   595→            if (chosen == null) {
   596→   596→                chosen = pages.firstOrNull { normalizedKey(it.title) == key }
   597→   597→            }
   598→   598→            if (chosen == null) {
   599→   599→                // try a sharper query "title film"
   600→   600→                chosen = wikiSearch("en", "$title film", 4)
   601→   601→                    .firstOrNull { filmDescRegex.containsMatchIn(it.description ?: "") }
   602→   602→            }
   603→   603→
   604→   604→            val wiki = chosen?.let { buildWikiEvidence(it) }
   605→   605→            val score = scoreCandidate(c, post, ocrText, wiki)
   606→   606→            if (wiki != null || score >= 0.5) {
   607→   607→                out.add(VerifiedCandidate(c, wiki, score))
   608→   608→            }
   609→   609→        }
   610→   610→        return out.sortedByDescending { it.score }
   611→   611→    }

   613→   613→    /* ================= self-built scoring ================= */
   614→   614→
   615→   615→    private fun normalizedKey(s: String): String =
   616→   616→        s.lowercase(Locale.ROOT).replace(Regex("[^a-z0-9\\p{L}\\p{N}]"), "")
   617→   617→
   618→   618→    private fun normalizedContains(haystack: String, needle: String): Boolean {
   619→   619→        val n = normalizedKey(needle)
   620→   620→        if (n.length < 3) return false
   621→   621→        return normalizedKey(haystack).contains(n)
   622→   622→    }
   623→   623→
   624→   624→    private fun scoreCandidate(
   625→   625→        c: Candidate,
   626→   626→        post: InstagramPost,
   627→   627→        ocrText: String,
   628→   628→        wiki: WikiEvidence?
   629→   629→    ): Double {
   630→   630→        var score = c.confidence.takeIf { it > 0.0 } ?: 0.5
   631→   631→        if (wiki != null && wiki.isFilm) score += 0.25
   632→   632→
   633→   633→        val title = c.name ?: return score.coerceIn(0.05, 0.97)
   634→   634→        val hay = listOfNotNull(post.caption, post.ogTitle, post.ownerUsername).joinToString(" ")
   635→   635→        if (normalizedContains(hay, title)) score += 0.10
   636→   636→        if (ocrText.isNotBlank() && normalizedContains(ocrText, title)) score += 0.15
   637→   637→        val tagHay = post.hashtags.joinToString("").lowercase(Locale.US)
   638→   638→        val titleKey = normalizedKey(title)
   639→   639→        if (titleKey.length >= 3 && tagHay.contains(titleKey)) score += 0.10
   640→   640→
   641→   641→        return score.coerceAtMost(0.97)
   642→   642→    }

   729→   729→                .append(" (").append(c.year ?: "?").append(") — ")

   746→   746→                if (!s.isNullOrBlank() && genres.size < 5) genres.add(s)
   747→   747→            }
   748→   748→        }

   750→   750→            found = optBool(obj, "found"),

   754→   754→            director = optString(obj, "director"),
   755→   755→            genres = genres,

   761→   761→    private fun blend(base: SmartResult, fused: Fused): SmartResult {
   762→   762→        if (!fused.found) {
   763→   763→            val capped = minOf(base.confidence, 0.55)
   764→   764→            return base.copy(
   765→   765→                confidence = capped,
   766→   766→                status = if (capped >= 0.55) "success" else "low_confidence"
   767→   767→            )
   768→   768→        }
   769→   769→        val sameTitle = normalizedKey(fused.titleEn ?: "").isNotEmpty() &&
   770→   770→            normalizedKey(fused.titleEn ?: "") == normalizedKey(base.titleEn ?: "")
   771→   771→        return if (sameTitle) {
   772→   772→            val conf = (0.6 * base.confidence + 0.4 * fused.confidence).coerceIn(0.0, 0.97)
   773→   773→            base.copy(
   774→   774→                confidence = conf,
   775→   775→                titleFa = base.titleFa ?: fused.titleFa,
   776→   776→                year = base.year ?: fused.year,
   777→   777→                director = base.director ?: fused.director,
   778→   778→                genres = if (fused.genres.isNotEmpty()) fused.genres else base.genres,
   779→   779→                reasoningFa = listOfNotNull(base.reasoningFa, fused.reasoningFa)
   780→   780→                    .joinToString(" — "),
   781→   781→                status = if (conf >= 0.55) "success" else "low_confidence"
   782→   782→            )
   783→   783→        } else if (fused.confidence > base.confidence) {
   784→   784→            val conf = (fused.confidence * 0.85).coerceIn(0.0, 0.9)
   785→   785→            SmartResult(
   786→   786→                status = if (conf >= 0.55) "success" else "low_confidence",
   787→   787→                titleEn = fused.titleEn,
   788→   788→                titleFa = fused.titleFa,
   789→   789→                year = fused.year ?: base.year,
   790→   790→                director = fused.director,
   791→   791→                genres = fused.genres,
   792→   792→                confidence = conf,
   793→   793→                reasoningFa = fused.reasoningFa,
   794→   794→                posterUrl = base.posterUrl,
   795→   795→                wikiUrl = base.wikiUrl,
   796→   796→                googleQuery = googleQueryFor(fused.titleEn, fused.year),
   797→   797→                candidates = base.candidates,
   798→   798→                sceneSummary = base.sceneSummary,
   799→   799→                evidenceSummary = base.evidenceSummary,
   800→   800→                instagramUrl = base.instagramUrl,
   801→   801→                sourceMode = base.sourceMode
   802→   802→            )
   803→   803→        } else {
   804→   804→            val capped = minOf(base.confidence, 0.7)
   805→   805→            base.copy(
   806→   806→                confidence = capped,
   807→   807→                status = if (capped >= 0.55) "success" else "low_confidence"
   808→   808→            )
   809→   809→        }
   810→   810→    }
